507 def compute_born(y_pred, sigma, alpha, mode="sigma"):
508 """Compute y_upper and y_lower boundary from gaussian hypothesis (sigma or 2sigma)
509  
510 Args:
511 y_pred (array) : Mean prediction
512 sigma (array) : Variance estimation
513 alpha (float) : Misscoverage ratio
514 mode (str) : Distribution hypothesis
515 (sigma : gaussian residual hypothesis, 2sigma : gaussian positive and negative residual hypothesis)
516  
517 Returns:
518 (y_lower,y_upper): Lower and upper bondary of Predictive interval
519 """
520 # Case sigma
521 if mode == "sigma":
522 y_lower = y_pred + scipy.stats.norm.ppf((alpha / 2), 0, sigma)
523 y_upper = y_pred + scipy.stats.norm.ppf((1 - (alpha / 2)), 0, sigma)
524 # Case 2 sigma
525 elif mode == "2sigma":
526 sigma = np.moveaxis(sigma, -1, 0)
527 y_lower = y_pred + scipy.stats.norm.ppf((alpha / 2), 0, sigma[0])
528 y_upper = y_pred + scipy.stats.norm.ppf((1 - (alpha / 2)), 0, sigma[1])
529 else:
530 raise NotImplementedError("This mode is not yet implemented.")
531 return y_lower, y_upper

717 self.fcorr = np.quantile(
718 residuals, (1 - self.alpha) * (1 + 1 / len(residuals))
719 )
720  
721 else:
722 self.fcorr_lower = np.quantile(
723 residuals[flag_res_lower],
724 np.minimum((1 - self.alpha) * (1 + 1 / flag_res_lower.sum()), 1),
725 )
726  
727 self.fcorr_upper = np.quantile(
728 residuals[flag_res_upper],
729 np.minimum((1 - self.alpha) * (1 + 1 / flag_res_upper.sum()), 1),
730 )
731 return

733 def calibrate(self, y_pred, y_pred_lower, y_pred_upper, sigma_pred, **kwargs):
734 if self.type_res == "res":
735 sigma_pred = y_pred * 0 + 1
736  
737 if self.mode == "symetric":
738 fcorr_lower = self.fcorr
739 fcorr_upper = self.fcorr
740 else:
741 fcorr_lower = self.fcorr_lower
742 fcorr_upper = self.fcorr_upper
743  
744 if self.type_res in ["res", "w_res", "no_calib"]:
745 if y_pred.shape != sigma_pred.shape:
746 sigma_pred = np.moveaxis(sigma_pred, -1, 0)
747 y_pred_lower = y_pred - sigma_pred[0] * fcorr_lower
748 y_pred_upper = y_pred + sigma_pred[1] * fcorr_upper
749 else:
750 y_pred_lower = y_pred - sigma_pred * fcorr_lower
751 y_pred_upper = y_pred + sigma_pred * fcorr_upper
752  
753 elif self.type_res in ["cqr"]:
754 y_pred_upper = y_pred_upper + fcorr_upper
755 y_pred_lower = y_pred_lower - fcorr_lower
756 else:
757 print("Unknown type_res")
758 return

791 def fit_compute_lag(Y, lag=[1, 2, 3], delay=0):
792 """Create lag features from a numerical array
793 Args:
794 Y (float array): Target to extract lag-feature
795 lag (int, optional): Lag number. Defaults to 3.
796 delay (int, optional): Delay before 1 lag feature. Defaults to 0.
797 """
798 dim = Y.shape
799 new_features_list = []
800 new_features_name = []
801 for i in np.array(lag) + delay:
802 Y_cur = np.roll(Y, i, axis=0)
803 if i > 0:
804 Y_cur[0:i] = 0
805 for g in range(dim[1]):
806 new_features_list.append(Y_cur[:, g])
807 new_features_name.append("lag_" + str(i) + "_dim:" + str(g))
808 new_features_name = np.array(new_features_name)
809 return (np.array(new_features_list).T, new_features_name)

818 def convolute_1D(array, filter=None):
819 """Convolution by dimension for 1 od 2D array using np.convolve
820  
821 Args:
822 array (_type_): array_to_convolve
823 filter (_type_, optional): convolution fitler. Defaults to None.
824  
825 Raises:
826 ValueError: dimension error
827  
828 Returns:
829 array: array_convoluted
830 """
831 if filter is None:
832 return array
833 else:
834 array_convoluted = []
835 if len(array.shape) == 1:
836 array_convoluted = np.convolve(array, filter, "same")
837 elif len(array.shape) == 2:
838 for i in range(array.shape[1]):
839 array_convoluted.append(
840 np.convolve(array[:, i], filter, "same")[:, None]
841 )
842 array_convoluted = np.concatenate(array_convoluted, axis=1)
843 else:
844 print("Inapropriate size of array : hold 1 or 2d array")
845 raise ValueError
846 return array_convoluted
